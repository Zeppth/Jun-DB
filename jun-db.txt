   6 |     NODE: 1,

  11 | export const JunCodec = {
  12 |     SIG: 0x4A554E,
  13 |     is(array) {
  14 |         return Array.isArray(array)
  15 |             && array[0] === this.SIG;
  16 |     },
  17 |     encode(type, data) {
  18 |         return [this.SIG, type, Date.now(),
  19 |         Buffer.from(data.toString()).toString('base64')]
  20 |     },
  21 |     decode(array) {
  22 |         if (array?.[0] !== this.SIG) return;
  23 |         return [array[1], array[2], Buffer.from(
  24 |             array[3], 'base64').toString('utf-8')]
  25 |     },
  26 | }
  27 | 
  28 | export class JunShard {
  29 |     constructor(JunDrive, JunMap, JunHub, depth) {
  30 |         this.JunDrive = JunDrive;
  31 |         this.JunMap = JunMap;
  32 |         this.JunHub = JunHub;
  33 |         this.depth = depth || 2;
  34 |     }
  35 | 
  36 |     static isObject(any) {
  37 |         if (!any) return false;
  38 |         if (typeof any !== 'object') return false;
  39 |         if (Array.isArray(any)) return false;
  40 |         const proto = Object.getPrototypeOf(any);
  41 |         return proto === Object.prototype
  42 |             || proto === null;
  43 |     }
  44 | 
  45 |     genId(depth) {
  46 |         depth = depth || this.depth;
  47 |         const id = crypto.randomBytes(6)
  48 |             .toString('hex').toUpperCase();
  49 |         if (!depth || depth <= 0) return `${id}`
  50 |         const folder = id.substring(0, depth);
  51 |         return `${folder}/${id}`;
  52 |     }
  53 | 
  54 |     forge(data, JunMap, JunHub) {
  55 |         if (!JunShard.isObject(data)) return;
  56 | 
  57 |         JunMap = JunMap || this.JunMap;
  58 |         JunHub = JunHub || this.JunHub;
  59 | 
  60 |         let mapContent = {};
  61 |         let hubContent = {};
  62 | 
  63 |         for (const key in data) {
  64 |             const value = data[key];
  65 | 
  66 |             if (JunShard.isObject(value)) {
  67 |                 const Id = this.genId();
  68 |                 const nodeFile = Id + '.node.bin';
  69 |                 const mapFile = Id + '.map.bin';
  70 | 
  71 |                 this.JunDrive.write(mapFile,
  72 |                     { $file: mapFile });
  73 | 
  74 |                 const _JunMap = new this.JunMap
  75 |                     .constructor(this.JunDrive, mapFile);
  76 | 
  77 |                 const _JunHub = new this.JunHub
  78 |                     .constructor(this.JunDrive, _JunMap);
  79 | 
  80 |                 this.forge(value, _JunMap, _JunHub);
  81 | 
  82 |                 hubContent[key] = JunCodec
  83 |                     .encode(1, nodeFile)
  84 |                 mapContent[key] = mapFile;
  85 |             } else {
  86 |                 hubContent[key] = value;
  87 |             }
  88 |         }
  89 | 
  90 |         Object.assign(JunMap
  91 |             .data, mapContent);
  92 |         Object.assign(JunHub
  93 |             .data, hubContent);
  94 | 
  95 |         JunMap.file.save();
  96 |         JunHub.file.save();
  97 |     }
  98 | 
  99 |     purge(mapPath) {
 100 |         if (!mapPath) return;
 101 |         if (typeof mapPath !== 'string') return;
 102 |         if (!mapPath.endsWith('.map.bin')) return;
 103 |         const mapData = this.JunDrive.readSync(mapPath);
 104 | 
 105 |         if (!mapData) return;
 106 |         for (const key in mapData) {
 107 |             if (key === '$file') continue;
 108 |             const childPath = mapData[key];
 109 |             this.purge(childPath);
 110 |         }
 111 | 
 112 |         const nodePath = mapPath.replace(
 113 |             '.map.bin', '.node.bin');
 114 | 
 115 |         this.JunDrive.remove(mapPath);
 116 |         this.JunDrive.remove(nodePath);
 117 |     }
